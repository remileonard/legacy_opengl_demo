# F-18 demo (Performer 2.2 and Coryphaeus EasyScene): the viewer of the
# Designer's Workbench files and the demo reuse the Performer layer of Mekton
set(PF_DIR ${CMAKE_SOURCE_DIR}/src/mekton)
set(F18_VIEW_SOURCES f18_view.c dwb.c dwb.h ${PF_DIR}/pf.c ${PF_DIR}/pfb.c)
set(F18_SOURCES f18.c escene.c escene.h dwb.c dwb.h ${PF_DIR}/pf.c ${PF_DIR}/pfb.c)

foreach(target f18_view f18)
    if(target STREQUAL "f18")
        set(sources ${F18_SOURCES})
    else()
        set(sources ${F18_VIEW_SOURCES})
    endif()
    if(NOT WIN32)
        add_executable(${target} ${sources})
    else()
        add_executable(${target} WIN32 ${sources})
        set_target_properties(${target} PROPERTIES LINK_FLAGS "/ENTRY:\"mainCRTStartup\"")
    endif()
    if (WIN32)
        target_link_libraries(${target} OpenGL::GL OpenGL::GLU $<IF:$<TARGET_EXISTS:FreeGLUT::freeglut>,FreeGLUT::freeglut,FreeGLUT::freeglut_static>)
    else()
        target_link_libraries(${target} $<IF:$<TARGET_EXISTS:FreeGLUT::freeglut>,FreeGLUT::freeglut,FreeGLUT::freeglut_static> OpenGL::GLU OpenGL::GL m)
    endif()
    target_include_directories(${target} PRIVATE ${PF_DIR} ${OPENGL_INCLUDE_DIRS} ${GLUT_INCLUDE_DIRS})
    target_compile_options(${target} PRIVATE ${OPENGL_CFLAGS} ${GLUT_CFLAGS})
    target_link_options(${target} PRIVATE ${OPENGL_LDFLAGS} ${GLUT_LDFLAGS})
endforeach()
