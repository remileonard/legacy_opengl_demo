if(NOT WIN32)
    add_executable(mekton_view mekton_view.c pfb.c pfb.h)
else()
    add_executable(mekton_view WIN32 mekton_view.c pfb.c pfb.h)
    set_target_properties(mekton_view PROPERTIES LINK_FLAGS "/ENTRY:\"mainCRTStartup\"")
endif()
if (WIN32)
    target_link_libraries(mekton_view OpenGL::GL OpenGL::GLU $<IF:$<TARGET_EXISTS:FreeGLUT::freeglut>,FreeGLUT::freeglut,FreeGLUT::freeglut_static>)
else()
    target_link_libraries(mekton_view $<IF:$<TARGET_EXISTS:FreeGLUT::freeglut>,FreeGLUT::freeglut,FreeGLUT::freeglut_static> OpenGL::GLU OpenGL::GL m)
endif()
target_include_directories(mekton_view PRIVATE ${OPENGL_INCLUDE_DIRS} ${GLUT_INCLUDE_DIRS})
target_compile_options(mekton_view PRIVATE ${OPENGL_CFLAGS} ${GLUT_CFLAGS})
target_link_options(mekton_view PRIVATE ${OPENGL_LDFLAGS} ${GLUT_LDFLAGS})
