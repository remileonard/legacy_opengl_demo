# Certain Impact: the viewer reuses the Performer layer of Mekton
set(PF_DIR ${CMAKE_SOURCE_DIR}/src/mekton)
set(CIMPACT_VIEW_SOURCES cimpact_view.c fst.c fst.h ${PF_DIR}/pf.c ${PF_DIR}/pfb.c)
if(NOT WIN32)
    add_executable(cimpact_view ${CIMPACT_VIEW_SOURCES})
else()
    add_executable(cimpact_view WIN32 ${CIMPACT_VIEW_SOURCES})
    set_target_properties(cimpact_view PROPERTIES LINK_FLAGS "/ENTRY:\"mainCRTStartup\"")
endif()
if (WIN32)
    target_link_libraries(cimpact_view OpenGL::GL OpenGL::GLU $<IF:$<TARGET_EXISTS:FreeGLUT::freeglut>,FreeGLUT::freeglut,FreeGLUT::freeglut_static>)
else()
    target_link_libraries(cimpact_view $<IF:$<TARGET_EXISTS:FreeGLUT::freeglut>,FreeGLUT::freeglut,FreeGLUT::freeglut_static> OpenGL::GLU OpenGL::GL m)
endif()
target_include_directories(cimpact_view PRIVATE ${PF_DIR} ${OPENGL_INCLUDE_DIRS} ${GLUT_INCLUDE_DIRS})
target_compile_options(cimpact_view PRIVATE ${OPENGL_CFLAGS} ${GLUT_CFLAGS})
target_link_options(cimpact_view PRIVATE ${OPENGL_LDFLAGS} ${GLUT_LDFLAGS})
